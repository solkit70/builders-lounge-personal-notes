🤔 왜 Multi-Agent인가?

단일 LLM 접근법의 한계

  • 컨텍스트 한계: 복잡한 문제는 토큰 제한 초과
  • 할루시네이션: 검증 없이 답변 생성
  • 단일 관점: 다양한 전문성 부재
  • 실행 불가: 생각만 하고 행동 못함
// 단일 LLM 접근법
prompt = "서버 CPU 95% 원인 분석해줘"
response = llm.call(prompt)
// → 추측만 가능, 실제 검증 불가

Multi-Agent 접근법

  • 역할 분담: 각 Agent가 전문 영역 담당
  • Tool 사용: 실제 시스템과 상호작용
  • 검증 루프: Agent 간 크로스체크
  • 확장성: 새 Agent 추가로 기능 확장
// Multi-Agent 접근법
Brain → "CPU 높음 감지"
  ↓
Judge → "심각도 평가, Investigation 요청"
  ↓
Investigation → ssh.top_processes 호출
  ↓
Specialist → "Java 프로세스 메모리 누수 발견"

}

AI Decision 구조

type AIDecision struct {
    Action          string   // 실행할 tool
    Reason          string   // 왜 이 tool인지
    Arguments       map[string]interface{}
    FinalHypothesis string   // 결론 (finish 시)
    Confidence      float64  // 확신도 (0~1)
}

// LLM에게 다음 행동 결정 요청
func DecideNextAction(state *Investigation) *AIDecision {
    prompt := buildPrompt(state)
    
    response := llm.Call(prompt)
    
    // JSON 파싱 (마크다운 백틱 제거 포함)
    content = strings.TrimPrefix(content, "```json")
    content = strings.TrimSuffix(content, "```")
    
    var decision AIDecision
    json.Unmarshal(content, &decision)
    return &decision
}

// LearningPipeline — 과거 인시던트에서 학습
type LearningPipeline struct {
    store   *IncidentStore  // Vector DB (Qdrant)
    records map[string]*LearningRecord
}

// Investigation 완료 시 자동 저장
func (lp *LearningPipeline) LearnFromInvestigation(
    record LearningRecord) error {
    
    // 1. 태그 자동 생성
    tags := lp.autoTag(record)
    
    // 2. Vector Store에 저장
    lp.store.Save(record)
    
    log.Printf("📚 Learned: %s → %s",

    return nil
}

// 새 조사 시 유사 사례 검색
func (lp *LearningPipeline) GetContextForInvestigation(
    goal, platform string, limit int) ([]ContextSuggestion, error) {
    
    // Vector Search로 유사 사례 찾기
    similar := lp.store.SearchSimilar(goal, limit)
    
    return similar, nil
}

// LLM 프롬프트에 주입
func (lp *LearningPipeline) FormatContextForPrompt(
    suggestions []ContextSuggestion) string {
    
    var sb strings.Builder
    sb.WriteString("\n📚 SIMILAR PAST INCIDENTS:\n")
    
    for _, s := range suggestions {
        sb.WriteString(fmt.Sprintf(
            "- [%s] %s → Root Cause: %s\n",
            s.IssueType, s.Situation, s.RootCause))
    }
    return sb.String()
}

// agents/ai_gate.go
type AIGate struct {

func (g *AIGate) Check(caller, detail string) GateResult {
    // 1. Emergency Stop
    if g.emergencyStop {
        return GateResult{Allowed: false, Reason: "Emergency stop"}
    }
    
    // 2. Rate Limit (분당 60회)
    if g.callsThisMinute >= g.maxCallsPerMinute {
        return GateResult{Allowed: false, Reason: "Rate limit"}
    }
    
    // 3. Cooldown (동일 문제 30초)
    sig := generateSignature(caller, detail)
    if lastCall, ok := g.cooldownMap[sig]; ok {
        if time.Since(lastCall) < g.cooldownPeriod {
            return GateResult{Allowed: false, Reason: "Cooldown"}
        }
    }
    
    // 4. Open Incident (이미 조사 중)
    if hasOpenIncident(sig) {
        return GateResult{Allowed: false, Reason: "Open incident"}
    }
    
    return GateResult{Allowed: true}
}

// db/postgres.go
type OpenIncident struct {
    ID         int64
    Signature  string    // SHA256 해시
    Status     string    // investigating/waiting/resolved
    Namespace  string
    Workload   string
    ProblemType string
    CreatedAt  time.Time
    UpdatedAt  time.Time
}

func generateSignature(ns, workload, problemType, msg string) string {
    data := fmt.Sprintf("%s|%s|%s|%s", 
        ns, workload, problemType, 
        normalizeMessage(msg))
    hash := sha256.Sum256([]byte(data))
    return hex.EncodeToString(hash[:])
}

// 숫자, 타임스탬프 정규화
func normalizeMessage(msg string) string {
    re := regexp.MustCompile(`\d+`)
    return re.ReplaceAllString(msg, "N")
}

// agents/registry.go
type AgentRegistry struct {
    mu      sync.RWMutex
    states  map[string]map[string]bool // parent → agent → enabled
}

func (r *AgentRegistry) IsEnabled(parent, agent string) bool {
    r.mu.RLock()
    defer r.mu.RUnlock()
    
    if agents, ok := r.states[parent]; ok {
        if enabled, ok := agents[agent]; ok {
            return enabled
        }
    }
    return true // default: enabled
}

// Agent Loop에서 체크
func (ji *JudgeInstance) runLoop(ctx context.Context) {

            time.Sleep(15 * time.Second)
            continue  // SKIP this cycle
        }
        
        ji.doWork()
        time.Sleep(15 * time.Second)
    }
}

LLM이 위험한 행동 시도     →  Human-in-the-Loop
Agent 추적 불가            →  AgentGroup 객체화
실행 병목                  →  TaskQueue + Worker Pool

JSON 파싱 문제

Haiku 문제: JSON을 마크다운으로 감쌈

// Haiku 응답
```json
{"action": "ssh.top_processes", ...}
```

// 파싱 에러!
invalid character '`'

해결: 백틱 제거 로직 추가

content = strings.TrimSpace(content)
if strings.HasPrefix(content, "```json") {
    content = strings.TrimPrefix(content, "```json")
}
if strings.HasSuffix(content, "```") {
    content = strings.TrimSuffix(content, "```")
}
content = strings.TrimSpace(content)
